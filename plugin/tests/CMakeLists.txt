# Unit tests for the pure parts of mxl2012-core (not in the plugin DLL). Included by ../CMakeLists.txt
# (cmake --build build/plugin --config Release --target tests), and buildable on its own without Windows headers or a
# resource compiler, as CI does on Linux: cmake -S plugin/tests -B build/tests.
cmake_minimum_required(VERSION 3.20)
project(mxl2012_core_tests LANGUAGES CXX)

set(names buildguard_tests otherplugins_tests envguard_tests harness_tests clientskills_tests missilestubs_tests itemeffect_tests skillremove_tests dodgelist_tests statelists_tests melttrail_tests whirlwind_tests origasserts_tests cubedupe_tests itemlevel_tests runeletters_tests skilltext_tests mercequip_tests dursync_tests soundguard_tests transguard_tests lvlupui_tests petspread_tests appleuse_tests vochat_tests)
add_executable(buildguard_tests EXCLUDE_FROM_ALL build_guard_test.cpp)
add_executable(otherplugins_tests EXCLUDE_FROM_ALL other_plugins_test.cpp)
add_executable(envguard_tests EXCLUDE_FROM_ALL env_guard_test.cpp)
add_executable(harness_tests EXCLUDE_FROM_ALL harness_cmds_test.cpp)
add_executable(clientskills_tests EXCLUDE_FROM_ALL client_skills_test.cpp)
add_executable(missilestubs_tests EXCLUDE_FROM_ALL missile_stubs_test.cpp)
add_executable(itemeffect_tests EXCLUDE_FROM_ALL item_effect_test.cpp)
add_executable(skillremove_tests EXCLUDE_FROM_ALL skill_remove_test.cpp)
add_executable(dodgelist_tests EXCLUDE_FROM_ALL dodge_list_test.cpp)
add_executable(statelists_tests EXCLUDE_FROM_ALL state_lists_test.cpp)
add_executable(melttrail_tests EXCLUDE_FROM_ALL melt_trail_test.cpp)
add_executable(whirlwind_tests EXCLUDE_FROM_ALL whirlwind_test.cpp)
add_executable(origasserts_tests EXCLUDE_FROM_ALL orig_asserts_test.cpp)
add_executable(cubedupe_tests EXCLUDE_FROM_ALL cube_dupe_test.cpp)
add_executable(itemlevel_tests EXCLUDE_FROM_ALL item_level_test.cpp)
add_executable(runeletters_tests EXCLUDE_FROM_ALL rune_letters_test.cpp)
add_executable(skilltext_tests EXCLUDE_FROM_ALL skill_text_test.cpp)
add_executable(mercequip_tests EXCLUDE_FROM_ALL merc_equip_test.cpp)
add_executable(dursync_tests EXCLUDE_FROM_ALL dur_sync_test.cpp)
add_executable(soundguard_tests EXCLUDE_FROM_ALL sound_guard_test.cpp)
add_executable(transguard_tests EXCLUDE_FROM_ALL translation_guard_test.cpp)
add_executable(lvlupui_tests EXCLUDE_FROM_ALL lvlup_ui_test.cpp)
add_executable(petspread_tests EXCLUDE_FROM_ALL pet_spread_test.cpp)
add_executable(appleuse_tests EXCLUDE_FROM_ALL apple_use_test.cpp)
add_executable(vochat_tests EXCLUDE_FROM_ALL vo_chat_test.cpp)
add_custom_target(tests DEPENDS ${names})

foreach(t ${names})
	target_compile_features(${t} PRIVATE cxx_std_20)
	if(MSVC)
		target_compile_options(${t} PRIVATE /W4 /permissive- /EHsc /utf-8)
	else()
		target_compile_options(${t} PRIVATE -Wall -Wextra)
	endif()
	if(NOT PROJECT_IS_TOP_LEVEL)
		# next to mxl2012-core.dll: build/plugin/<config>/<test>.exe
		set_target_properties(${t} PROPERTIES RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR})
	endif()
endforeach()
